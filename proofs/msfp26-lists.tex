\begin{theorem}
    The list object $L(A)$ is the free monoid on $A$.
\end{theorem}
\begin{proof}
    Define $\App:=\iter(\id,\cons):L(A)\otimes L(A)\to L(A)$. We claim that $(L(A),\nil,\App)$ is a free monoid structure on $A$. The left unit law holds by $\nil$-$\beta$:

                \draw[blue] (in1.east) to[out=0,in=140] node[inputwire,pos=0.4,above,sloped=false] {$L(A)$} (op.140);
                \draw[blue] ($(in2.west) - (0.25,0)$) to[out=0,in=220] node[inputwire,below,sloped=false] {$L(A)$} (op.220);
                \draw[blue] (op.east) -- (out.west) node[outputwire,above] {$L(A)$};
            \end{scope}

            \node at (2,0) {$=$};

            \begin{scope}[xshift=4cm]

            \end{scope}
          \end{tikzpicture}%
    \end{equation*}
    For the right unit law, consider the maps $f:=(-)\App\nil:L(A)\to L(A)$ and $g=\id:L(A)\to L(A)$:

\section{List objects}
\label{sec:list-objects}

Let $\cat{C}(1, \otimes)$ be a strict monoidal category.

\begin{definition}

  \[
    \Nullary{\nil}{L(A)} \quad \text{and} \quad \smash{\Binary{A}{L(A)}{\cons}{L(A)}}
  \]
  such that for every object $L$ equipped with maps
  \[
    \Unary{P}{n}{L} \quad \text{and} \quad \smash{\Binary{A}{L}{c}{L}}
  \]
  there exists a unique \textit{mediating map} or \textit{iterator}
  \[
    \smash{\Binary{L(A)}{P}{\iter(n,c)}{L}}

\begin{theorem}[Lemma~3.4.2 of \cite{fioreListObjectsAlgebraic2017}]
  If $\cat{C}$ admits parametrised list objects for every $A$,
  then $\mathsf{Mon}(\cat{C})$ is monadic over $\cat{C}$.
\end{theorem}



\begin{theorem}[Lemma~3.4.3 of \cite{fioreListObjectsAlgebraic2017}]
  If $(\cat{C},\otimes,1)$ is cartesian, then the induced monad $T:\cat{C}\to\cat{C}$ is strong.
\end{theorem}

\begin{theorem}
  The following are equivalent:
  \begin{enumerate}
    \item $(L(A),\nil,\cons,\mathrm{iter})$ is a symmetric list object;
    \item $(L(A),\nil,+\kern-0.4em+)$ is a commutative monoid object;
    \item There is a joint coequaliser of every permutation of symmetric powers $L(A)^{\otimes n}$
          \begin{center}
            \begin{tikzpicture}[scale=1.2]
              \node (l) at (0,0) {$L(A)^{\otimes n}$};
              \node (r) at (2,0) {$L(A)^{\otimes n}$};
              \draw[->,thin] ([yshift=1ex]l.east) -- node[midway,above] {$\sigma$} ([yshift=1ex]r.west);
              \draw[->,dashed,thin] (l) -- (r);
              \draw[->,thin] ([yshift=-1ex]l.east) -- ([yshift=-1ex]r.west);
            \end{tikzpicture}
          \end{center}
          for every $n\ge2$.
  \end{enumerate}
\end{theorem}

\begin{theorem}
  If $\cat{C}$ admits parametrised symmetric list objects for every $A$,
  then the forgetful functor $U:\mathsf{CMon}(\cat{C})\to\cat{C}$ is monadic.
\end{theorem}

\begin{theorem}
  If $(\cat{C},\otimes,1)$ is cartesian, then the induced monad $T:\cat{C}\to\cat{C}$ is strong, and further, commutative.
\end{theorem}
